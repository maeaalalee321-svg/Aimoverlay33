    // ------------------------------------------------------------ الكرة الهدف




    /**
     * يلتقط الخط الأبيض القصير الذي ترسمه اللعبة من مركز الكرة الهدف؛ هذا الخط يمرّ بمركز الكرة الهدف
     * وبمركز الكرة الشبحية، فنجمع البكسلات البيضاء الصارمة حول الشبحية ونبحث عن أكبر عنقود مستقيم
     * ثم نلاءمه بـ PCA متكرر. أدق بكثير من تقدير مركز الكرة بلونها.
     * يرجع [nx, ny, ox, oy] أو null.
     */
    private fun findTick(
        px: IntArray, w: Int, h: Int, r: Float, gx: Float, gy: Float, ux: Float, uy: Float
    ): FloatArray? {
        dbg = ""
        val nb = 180
        val hist = IntArray(nb)
        val xs = ArrayList<Int>()
        val ys = ArrayList<Int>()
        val ya = max(0, (gy - 5f * r).toInt()); val yb = min(h - 1, (gy + 5f * r).toInt())
        val xa = max(0, (gx - 5f * r).toInt()); val xb = min(w - 1, (gx + 5f * r).toInt())
        val twoPi = (2.0 * Math.PI).toFloat()
        for (y in ya..yb) for (x in xa..xb) {
            val vx = x - gx; val vy = y - gy
            val d2 = vx * vx + vy * vy
            if (d2 < 3.42f * r * r || d2 > 25f * r * r) continue
            if (!isWhiteStrict(px[y * w + x])) continue
            val d = sqrt(d2)
            if ((vx * ux + vy * uy) / d < -0.2f) continue          // خلف الشبحية: خط الكرة البيضاء
            var bi = (((atan2(vy, vx) + Math.PI.toFloat()) / twoPi) * nb).toInt()
            if (bi >= nb) bi = nb - 1
            if (bi < 0) bi = 0
            hist[bi]++
            xs.add(x); ys.add(y)
        }
        val total = xs.size
        if (total < 12) return null
        val sm = IntArray(nb) { hist[it] + hist[(it + 1) % nb] + hist[(it + nb - 1) % nb] }
        val used = ArrayList<Int>()
        var bestN = 0
        var res: FloatArray? = null
        val band = BooleanArray(total)
        for (cand in 0 until 4) {
            var bi = -1
            var bv = 7
            for (i in 0 until nb) {
                if (sm[i] <= bv) continue
                var near = false
                for (u in used) { val dd = abs(i - u); if (min(dd, nb - dd) <= 3) { near = true; break } }
                if (near) continue
                bv = sm[i]; bi = i
            }
            if (bi < 0) break
            used.add(bi)
            val phi = -Math.PI.toFloat() + (bi + 0.5f) * twoPi / nb
            // بين حلقة الشبحية ومركز الهدف لا يوجد أبيض في خط الهدف، بخلاف خط انحراف الكرة البيضاء
            var c = 0
            var tot = 0
            var t = 1.3f * r
            while (t <= 1.9f * r) {
                val xx = (gx + cos(phi) * t + 0.5f).toInt()
                val yy = (gy + sin(phi) * t + 0.5f).toInt()
                tot++
                if (xx in 0 until w && yy in 0 until h && isWhiteStrict(px[yy * w + xx])) c++
                t += 1f
            }
            if (tot > 0 && c.toFloat() / tot >= 0.4f) continue

            var cur = phi
            var mx = gx
            var my = gy
            var n = 0
            var ok = true
            for (iter in 0 until 3) {
                val dx0 = cos(cur); val dy0 = sin(cur)
                var sx = 0.0; var sy = 0.0; var k = 0
                for (i in 0 until total) {
                    val vx = xs[i] - mx; val vy = ys[i] - my
                    val inb = vx * dx0 + vy * dy0 > -2f * r && abs(vx * dy0 - vy * dx0) <= 2.8f
                    band[i] = inb
                    if (inb) { sx += xs[i]; sy += ys[i]; k++ }
                }
                if (k < 8) { ok = false; break }
                val mxx = sx / k; val myy = sy / k
                var sxx = 0.0; var syy = 0.0; var sxy = 0.0
                for (i in 0 until total) if (band[i]) {
                    val ddx = xs[i] - mxx; val ddy = ys[i] - myy
                    sxx += ddx * ddx; syy += ddy * ddy; sxy += ddx * ddy
                }
                var th = (0.5 * atan2(2.0 * sxy, sxx - syy)).toFloat()
                if (cos(th) * dx0 + sin(th) * dy0 < 0f) th += Math.PI.toFloat()
                cur = atan2(sin(th), cos(th))
                mx = mxx.toFloat(); my = myy.toFloat()
                n = k
            }
            if (!ok) continue
            val dxf = cos(cur); val dyf = sin(cur)
            var amin = Float.MAX_VALUE
            var amax = -Float.MAX_VALUE
            for (i in 0 until total) {
                val vx = xs[i] - mx; val vy = ys[i] - my
                if (vx * dxf + vy * dyf > -2f * r && abs(vx * dyf - vy * dxf) <= 2.8f) {
                    val al = (xs[i] - gx) * dxf + (ys[i] - gy) * dyf
                    if (al < amin) amin = al
                    if (al > amax) amax = al
                }
            }
            dbg += " [n=$n ext=${((amax - amin) / r * 10).toInt() / 10f} ang=${Math.toDegrees(cur.toDouble()).toInt()} fwd=${dxf * ux + dyf * uy}]"
            if (n < 24 || amax - amin < r) continue
            if (dxf * ux + dyf * uy < 0.02f) continue             // الكرة الهدف تتحرك للأمام دائماً
            if (n > bestN) {
                bestN = n
                val tt = (gx + 2f * r * dxf - mx) * dxf + (gy + 2f * r * dyf - my) * dyf
                res = floatArrayOf(dxf, dyf, mx + dxf * tt, my + dyf * tt)
            }
        }
        return res
    }

    /**
     * مركز الكرة الشبحية بدقة: الحلقة البيضاء دائرة نصف قطرها r. نبحث على امتداد خط التصويب
     * عن الموضع الذي تقع عنده أكثر نقاط الحلقة على بكسلات بيضاء. (نهاية الخط الأبيض وحدها
     * تتجاوز الحلقة أحياناً بنصف قطر أو أكثر فتُخطئ اتجاه الكرة الهدف.)
     * يرجع المسافة من مركز الكرة البيضاء إلى مركز الحلقة، أو g0 إن لم تُوجد.
     */
    private fun fitGhost(
        px: IntArray, w: Int, h: Int, cx: Float, cy: Float, ux: Float, uy: Float, r: Float, g0: Float
    ): Float {
        val ns = 36
        val cs = FloatArray(ns) { cos(it * 2f * Math.PI.toFloat() / ns) * r }
        val sn = FloatArray(ns) { sin(it * 2f * Math.PI.toFloat() / ns) * r }
        var best = -1
        var t = g0 - 3.2f * r
        val ts = ArrayList<Float>()
        val cnt = ArrayList<Int>()
        while (t <= g0 + 0.6f * r) {
            if (t > 2.2f * r) {
                val gx = cx + ux * t; val gy = cy + uy * t
                var c = 0
                for (j in 0 until ns) {
                    val xx = (gx + cs[j] + 0.5f).toInt(); val yy = (gy + sn[j] + 0.5f).toInt()
                    var hit = false
                    for (oy in -1..1) {
                        for (ox in -1..1) {
                            val x = xx + ox; val y = yy + oy
                            if (x in 0 until w && y in 0 until h && isWhiteStrict(px[y * w + x])) { hit = true; break }
                        }
                        if (hit) break
                    }
                    if (hit) c++
                }
                ts.add(t); cnt.add(c)
                if (c > best) best = c
            }
            t += 0.5f
        }
        if (best < 22) return g0
        var sum = 0f
        var n = 0
        for (i in ts.indices) if (cnt[i] >= best - 1) { sum += ts[i]; n++ }
        return if (n > 0) sum / n else g0
    }

    /**
     * مطابقة قالب: الكرة الهدف قرص نصف قطره r مركزه على بعد 2r من مركز الحلقة في اتجاه φ.
     * نختار φ التي يحوي قرصها أكثر بكسلات «كرة» وأقل بكسلات قماش. مستقلة عن تحيّز الحواف.
     * يرجع [cosφ, sinφ, score] أو null.
     */
    private fun templateContact(
        px: IntArray, w: Int, h: Int, tb: FloatArray, r: Float, gx: Float, gy: Float, ux: Float, uy: Float
    ): FloatArray? {
        val base = atan2(uy, ux)
        val rr = 0.9f * r
        val skip2 = 1.35f * r * 1.35f * r
        val sc = FloatArray(181)
        var best = -2f
        var bi = 0
        for (k in 0..180) {
            val phi = base + (k - 90) * 0.0174533f
            val cx = gx + 2f * r * cos(phi)
            val cy = gy + 2f * r * sin(phi)
            // مركز كرة لا يمكن أن يقع في حزام الحافة (الظل والميل يشبهان الكرة بلونهما)
            if (cx < tb[0] + 2.55f * r || cx > tb[2] - 2.55f * r || cy < tb[1] + 2.55f * r || cy > tb[3] - 2.55f * r) {
                sc[k] = -2f
                continue
            }
            val y0 = max(0, (cy - rr).toInt()); val y1 = min(h - 1, (cy + rr).toInt() + 1)
            val x0 = max(0, (cx - rr).toInt()); val x1 = min(w - 1, (cx + rr).toInt() + 1)
            var sum = 0
            var n = 0
            for (y in y0..y1) for (x in x0..x1) {
                val dx = x - cx; val dy = y - cy
                if (dx * dx + dy * dy > rr * rr) continue
                val ex = x - gx; val ey = y - gy
                if (ex * ex + ey * ey < skip2) continue
                if (x < tb[0] + 1.55f * r || x > tb[2] - 1.55f * r || y < tb[1] + 1.55f * r || y > tb[3] - 1.55f * r) continue   // وجه الحافة
                val c = px[y * w + x]
                n++
                sum += if (isFelt(c)) -1 else if (isWhite(c) || isOverlayColor(c) || nearExcl(x.toFloat(), y.toFloat())) 0 else 1
            }
            val v = if (n < 40) -2f else sum.toFloat() / n
            sc[k] = v
            if (v > best) { best = v; bi = k }
        }
        if (best < 0.15f) return null
        var sx = 0.0; var sy = 0.0
        for (k in max(0, bi - 25)..min(180, bi + 25)) if (sc[k] >= best - 0.03f) {
            val phi = base + (k - 90) * 0.0174533f
            sx += cos(phi); sy += sin(phi)
        }
        val a = atan2(sy, sx).toFloat()
        return floatArrayOf(cos(a), sin(a), best)
    }



    // ------------------------------------------------------------ الحلقة وخط الكرة الهدف (v12)

    /** هل نقطة الحلقة تقع داخل الكرة البيضاء (تُهمل لأنها بيضاء أصلاً)؟ */
    private fun inCue(x: Float, y: Float, cx: Float, cy: Float, r: Float): Boolean {
        val dx = x - cx; val dy = y - cy
        return dx * dx + dy * dy < 1.1f * r * 1.1f * r
    }

    /** نقاط حلقة نصف قطرها rad حول (gx,gy): تعيد [عدد_صالح, عدد_أبيض] مع تسامح بكسل. */
    private fun ringCount(
        px: IntArray, w: Int, h: Int, gx: Float, gy: Float, rad: Float, ns: Int, cx: Float, cy: Float, r: Float
    ): IntArray {
        var valid = 0
        var white = 0
        for (j in 0 until ns) {
            val a = j * 2f * Math.PI.toFloat() / ns
            val sx = gx + cos(a) * rad
            val sy = gy + sin(a) * rad
            if (inCue(sx, sy, cx, cy, r)) continue
            val xx = (sx + 0.5f).toInt(); val yy = (sy + 0.5f).toInt()
            if (xx < 1 || yy < 1 || xx >= w - 1 || yy >= h - 1) continue
            valid++
            var hit = false
            for (oy in -1..1) {
                for (ox in -1..1) if (isWhiteStrict(px[(yy + oy) * w + xx + ox])) { hit = true; break }
                if (hit) break
            }
            if (hit) white++
        }
        return intArrayOf(valid, white)
    }

    /** الحلقة ليست قرصاً أبيض ممتلئاً: داخلها (نصف القطر 0.5) يجب أن يكون في معظمه ليس أبيض. */
    private fun ringHollow(px: IntArray, w: Int, h: Int, gx: Float, gy: Float, cx: Float, cy: Float, r: Float): Boolean {
        var v = 0
        var wh = 0
        for (j in 0 until 16) {
            val a = j * 2f * Math.PI.toFloat() / 16
            val sx = gx + cos(a) * 0.5f * r
            val sy = gy + sin(a) * 0.5f * r
            if (inCue(sx, sy, cx, cy, r)) continue
            val xx = (sx + 0.5f).toInt(); val yy = (sy + 0.5f).toInt()
            if (xx < 0 || yy < 0 || xx >= w || yy >= h) continue
            v++
            if (isWhiteStrict(px[yy * w + xx])) wh++
        }
        return v < 5 || wh <= 0.45f * v
    }

    /**
     * بحث ثنائي الأبعاد عن حلقة الكرة الشبحية قرب الكرة البيضاء (الضربات القريبة حيث لا يظهر خط التصويب).
     * يرجع [x, y] مركز الحلقة أو null.
     */
    private fun findRingNear(px: IntArray, w: Int, h: Int, cx: Float, cy: Float, r: Float): FloatArray? {
        val lim = 6f * r
        val ns = 48
        var best = -1000
        val cxs = ArrayList<Float>(); val cys = ArrayList<Float>(); val scs = ArrayList<Int>()
        val x0 = max(2, (cx - lim).toInt()); val x1 = min(w - 3, (cx + lim).toInt())
        val y0 = max(2, (cy - lim).toInt()); val y1 = min(h - 3, (cy + lim).toInt())
        for (y in y0..y1) for (x in x0..x1) {
            val dx = x - cx; val dy = y - cy
            val d2 = dx * dx + dy * dy
            if (d2 > lim * lim || d2 < 0.3f * r * 0.3f * r) continue
            val gx = x.toFloat(); val gy = y.toFloat()
            // اختبار سريع: مركز الحلقة ليس أبيض (إلا خطوط التصويب) — نتخطى فقط إن كان قرصاً أبيض ممتلئاً
            val qc = ringCount(px, w, h, gx, gy, r, 8, cx, cy, r)
            if (qc[0] < 3 || qc[1] < 0.5f * qc[0]) continue
            val rc = ringCount(px, w, h, gx, gy, r, ns, cx, cy, r)
            if (rc[0] < 22) continue
            if (rc[1] < 0.68f * rc[0]) continue
            if (!ringHollow(px, w, h, gx, gy, cx, cy, r)) continue
            val sc = rc[1] * 2 - rc[0]
            if (sc > best) best = sc
            cxs.add(gx); cys.add(gy); scs.add(sc)
        }
        if (best < 0) return null
        var sx = 0.0; var sy = 0.0; var n = 0
        for (i in scs.indices) if (scs[i] >= best - 3) { sx += cxs[i]; sy += cys[i]; n++ }
        return floatArrayOf((sx / n).toFloat(), (sy / n).toFloat())
    }
